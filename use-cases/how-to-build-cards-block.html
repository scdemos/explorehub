How to Build a Cards Block

Flexible card layouts with three variants: default, links, and bento

Block Query Index Intermediate

Overview

The Cards block displays content in flexible card layouts with three distinct variants: authored cards (default), query-index powered cards (links), and asymmetric bento grids (bento). Each variant provides different authoring patterns and visual layouts.

Key Features

  • Three variants - Default, Links, Bento for different use cases
  • Query-index integration - Dynamic content from site metadata
  • Image optimization - Automatic responsive image generation
  • Link wrapping - Entire card clickable in default variant
  • Bento featured card - First card highlighted in grid
  • Universal Editor support - Preserves authored links

Step 3: JavaScript Decoration

The blocks/cards/cards.js file adds semantic classes and creates proper card structure:

export default function decorate(block) {
  const ul = document.createElement('ul');
  [...block.children].forEach((row) => {
    const li = document.createElement('li');
    li.className = 'cards-card';  // ← Adds card class

    while (row.firstElementChild) {
      li.append(row.firstElementChild);
    }

    [...li.children].forEach((div) => {
      if (div.children.length === 1 && div.querySelector('picture')) {
        div.className = 'cards-card-image';  // ← Marks image div
      } else {
        div.className = 'cards-card-body';   // ← Marks text div
      }
    });

    ul.append(li);
  });

  block.textContent = '';
  block.append(ul);
}

How It Works

Step 1: Detect Variant

The main decorate() function checks CSS classes to determine variant (cards.js:183-188):

export default async function decorate(block) {
  if (block.classList.contains('links')) {
    await decorateLinks(block);
  } else if (block.classList.contains('bento')) {
    decorateBento(block);
  } else {
    decorateDefault(block);
  }
}

Step 2: Links Variant - Query Index

For links variant, extract paths and resolve metadata (cards.js:36-56):

async function decorateLinks(block) {
  // Extract authored link paths
  const authoredLinks = getAuthoredLinks(block);

  // Fetch site-wide metadata index
  const indexRows = await fetchQueryIndexAll();

  // Resolve article metadata for each path
  const articles = resolveArticlesFromIndex(authoredLinks, indexRows);

  // Build card for each article
  const ul = createTag('ul');
  articles.forEach((article) => {
    ul.append(buildLinksCard(article));
  });

  block.replaceChildren(ul);
}

Step 3: Build Links Card

Create card from article metadata (cards.js:11-31):

function buildLinksCard(article) {
  const href = normalizePath(article.path);
  const li = createTag('li');
  const link = createTag('a', { href, class: 'cards-card-link' });

  // Add optimized image if available
  if (article.image) {
    const imageDiv = createTag('div', { class: 'cards-card-image' });
    imageDiv.append(
      createOptimizedPicture(article.image, article.title, false, [{ width: '750' }])
    );
    link.append(imageDiv);
  }

  // Add title and description
  const body = createTag('div', { class: 'cards-card-body' });
  body.append(createTag('p', {}, createTag('strong', {}, article.title)));
  if (article.description) {
    body.append(createTag('p', {}, article.description));
  }

  link.append(body);
  li.append(link);
  return li;
}

Step 4: Default Variant - Card Wrapping

For default variant, wrap content in clickable link (cards.js:115-180):

function decorateDefault(block) {
  const ul = createTag('ul');

  [...block.children].forEach((row) => {
    const li = createTag('li');

    // Extract image and body content
    const imageDiv = createTag('div', { class: 'cards-card-image' });
    const bodyDiv = createTag('div', { class: 'cards-card-body' });

    // Find link in authored content
    const linkEl = li.querySelector('a[href]');

    if (linkEl) {
      // Wrap entire card in link
      const wrapper = createTag('a', {
        href: linkEl.getAttribute('href'),
        class: 'cards-card-link',
      });
      while (li.firstChild) wrapper.append(li.firstChild);
      li.append(wrapper);

      // Remove original link element
      linkEl.replaceWith(...linkEl.childNodes);
    }

    ul.append(li);
  });

  // Optimize all images
  ul.querySelectorAll('picture > img').forEach((img) => {
    img.closest('picture').replaceWith(
      createOptimizedPicture(img.src, img.alt, false, [{ width: '750' }])
    );
  });

  block.replaceChildren(ul);
}

Step 5: Bento Variant - Featured Card

For bento variant, mark first card as featured and extract tags (cards.js:64-110):

function decorateBento(block) {
  const ul = createTag('ul');

  [...block.children].forEach((row, idx) => {
    const li = createTag('li');

    // First card is featured
    if (idx === 0) {
      li.classList.add('cards-card-featured');
    }

    // Extract image into wrapper
    const picture = li.querySelector('picture');
    if (picture) {
      const imageDiv = createTag('div', { class: 'cards-card-image' });
      imageDiv.append(picture);
      li.prepend(imageDiv);
    }

    // Mark first 
 as tag/label
    const firstP = li.querySelector('p');
    if (firstP) {
      firstP.classList.add('cards-card-tag');
    }

    // Wrap remaining content in body
    const body = createTag('div', { class: 'cards-card-body' });
    [...li.children].forEach((child) => {
      if (!child.classList.contains('cards-card-image')) {
        body.append(child);
      });
    li.append(body);

    ul.append(li);
  });

  block.replaceChildren(ul);
}
