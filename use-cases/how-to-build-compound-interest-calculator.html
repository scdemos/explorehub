How to Build a Compound Interest Calculator Block

Interactive financial calculator with Chart.js visualization and real-time calculations

Block Interactive Advanced

Overview

The Compound Interest Calculator is an interactive financial planning tool that helps users project investment growth over time. It calculates compound interest with periodic contributions, displays results with a stacked area chart using Chart.js, and provides optional yearly breakdowns.

Key Features:

  • Real-time calculation as users adjust inputs
  • Configurable initial investment and regular contributions (monthly, biweekly, weekly, annually)
  • Multiple compounding frequencies (daily, monthly, annually)
  • Optional inflation adjustment to show "real" purchasing power
  • Stacked area chart visualization showing investment value vs. interest earned
  • Optional yearly breakdown table with sortable columns
  • Responsive layout with form on left and results on right (desktop)
  • Accessible with ARIA live regions and keyboard support
  • CAD currency formatting (easily customizable for other locales)

How It Works

The calculator follows this workflow:

graph TD A[User Adjusts Inputs] --> B[Read Form Values] B --> C[Calculate Monthly Compounding] C --> D[Build Data Points] D --> E{Show Breakdown?} E -->|Yes| F[Render Yearly Table] E -->|No| G[Load Chart.js] G --> H[Render Stacked Area Chart] F --> I[Update Output ARIA Live] H --> I I --> J[Display Metrics] J --> K[User Changes Input] K --> A

Calculation Logic:

  1. Convert to Monthly Rate: Convert annual rate and compounding frequency to effective monthly rate
  2. Monthly Simulation: Iterate month-by-month, applying interest and adding contributions
  3. Track Balance: Keep cumulative balance, contributions, and interest
  4. Inflation Adjustment: If enabled, divide final balance by (1 + inflation)^years to get "real" value
  5. Chart Data: Sample data points (max 120 points) for Chart.js visualization
  6. Yearly Snapshots: If breakdown enabled, capture balance at each year-end

import { createTag, loadChartJs, createChart } from '../../scripts/shared.js';

const DEFAULTS = {
  initialInvestment: 10000,
  regularContribution: 500,
  contributionFrequency: 'monthly',
  annualRate: 5,
  compoundingFrequency: 'monthly',
  years: 10,
  adjustForInflation: false,
  inflationRate: 2,
  showYearlyBreakdown: false,
};

const CONTRIB_FREQ_PER_YEAR = {
  monthly: 12, biweekly: 26, weekly: 52, annually: 1,
};
const COMPOUND_PER_YEAR = {
  annually: 1, monthly: 12, daily: 365,
};

function toNumber(value, fallback = 0) {
  const parsed = Number(value);
  return Number.isFinite(parsed) ? parsed : fallback;
}

function money(value) {
  return new Intl.NumberFormat('en-CA', {
    style: 'currency',
    currency: 'CAD',
    maximumFractionDigits: 0,
  }).format(value || 0);
}

function moneyDecimals(value, decimals = 2) {
  return new Intl.NumberFormat('en-CA', {
    style: 'currency',
    currency: 'CAD',
    minimumFractionDigits: decimals,
    maximumFractionDigits: decimals,
  }).format(value ?? 0);
}

// Compound interest calculation with periodic contributions
function calculate(values) {
  const initial = Math.max(0, toNumber(values.initialInvestment, DEFAULTS.initialInvestment));
  const contribution = Math.max(0, toNumber(values.regularContribution, DEFAULTS.regularContribution));
  const contribFreq = values.contributionFrequency || DEFAULTS.contributionFrequency;
  const annualRate = toNumber(values.annualRate, DEFAULTS.annualRate) / 100;
  const compoundFreq = values.compoundingFrequency || DEFAULTS.compoundingFrequency;
  const years = Math.max(0.5, Math.min(50, toNumber(values.years, DEFAULTS.years)));
  const adjustInflation = Boolean(values.adjustForInflation);
  const inflationRate = toNumber(values.inflationRate, DEFAULTS.inflationRate) / 100;
  const showBreakdown = Boolean(values.showYearlyBreakdown);

  const periodsPerYear = COMPOUND_PER_YEAR[compoundFreq] ?? 12;
  const monthlyRate = periodsPerYear === 1
    ? (1 + annualRate) ** (1 / 12) - 1
    : (1 + annualRate / periodsPerYear) ** (periodsPerYear / 12) - 1;

  const contribsPerYear = CONTRIB_FREQ_PER_YEAR[contribFreq] ?? 12;
  const contributionPerMonth = (contribution * contribsPerYear) / 12;

  let balance = initial;
  let cumulativeContrib = initial;
  const monthsTotal = Math.round(years * 12);
  const yearlySnapshots = showBreakdown ? [] : null;
  const maxChartPoints = 120;
  const chartStep = Math.max(1, Math.floor(monthsTotal / maxChartPoints));
  const chartData = [];
  const contributionsPerYear = 12 * contributionPerMonth;

  for (let m = 0; m < monthsTotal; m += 1) {
    balance = balance * (1 + monthlyRate) + contributionPerMonth;
    cumulativeContrib += contributionPerMonth;
    if (showBreakdown && (m + 1) % 12 === 0) {
      yearlySnapshots.push({
        year: (m + 1) / 12,
        balance,
        totalInvestment: cumulativeContrib,
      });
    }
    if (m % chartStep === 0 || m === monthsTotal - 1) {
      chartData.push({
        month: m + 1,
        valueOfInvestment: cumulativeContrib,
        totalValue: balance,
        interestEarned: balance - cumulativeContrib,
      });
    }
  }

  const totalContributions = cumulativeContrib;
  const totalInterest = balance - totalContributions;
  const realFinalBalance = adjustInflation && inflationRate > 0
    ? balance / (1 + inflationRate) ** years
    : null;

  return {
    finalBalance: balance,
    totalContributions,
    totalInterest,
    realFinalBalance,
    yearlyBreakdown: yearlySnapshots,
    yearlyBreakdownInitial: initial,
    yearlyBreakdownContributionsPerYear: contributionsPerYear,
    chartData,
    monthsTotal,
    years,
  };
}

// Build and render Chart.js stacked area chart
function drawChartWithChartJs(canvas, chartData) {
  if (!chartData.length || !canvas) return undefined;
  const labels = chartData.map((d) => d.month);
  const valueOfInvestmentData = chartData.map((d) => d.valueOfInvestment);
  const interestData = chartData.map((d) => d.totalValue - d.valueOfInvestment);
  const config = {
    type: 'line',
    data: {
      labels,
      datasets: [
        {
          label: 'Value of investment',
          data: valueOfInvestmentData,
          backgroundColor: 'rgba(184, 217, 235, 0.7)',
          borderColor: 'rgb(126, 184, 218)',
          borderWidth: 0,
          fill: true,
          tension: 0.2,
        },
        {
          label: 'Total interest earned',
          data: interestData,
          backgroundColor: 'rgba(57, 96, 119, 0.7)',
          borderColor: 'rgb(90, 138, 158)',
          borderWidth: 0,
          fill: true,
          tension: 0.2,
        },
      ],
    },
    options: {
      responsive: true,
      maintainAspectRatio: true,
      aspectRatio: 2.3,
      interaction: { intersect: false, mode: 'index' },
      plugins: {
        legend: { display: true, position: 'top' },
        tooltip: {
          callbacks: {
            label(ctx) {
              const v = ctx.raw;
              return \`\${ctx.dataset.label}: \${money(v)}\`;
            },
          },
        },
      },
      scales: {
        x: {
          title: { display: true, text: 'Months' },
          stacked: true,
        },
        y: {
          title: { display: true, text: 'Total value' },
          stacked: true,
          beginAtZero: true,
          ticks: {
            callback(value) {
              return money(value);
            },
          },
        },
      },
    },
  };
  return createChart(canvas, config);
}

Chart.js Integration

The calculator uses shared Chart.js utilities from scripts/shared.js to keep Chart.js off the critical loading path:

// shared.js exports loadChartJs() and createChart()
import { loadChartJs, createChart } from '../../scripts/shared.js';

// Lazy-load Chart.js when chart container is built
loadChartJs()
  .then(() => {
    const chart = drawChartWithChartJs(canvas, chartData);
    block.compoundInterestChart = chart; // Store for cleanup on re-render
  })
  .catch(() => {
    // Fallback message if Chart.js fails to load
    wrap.append(createTag('p', { class: 'chart-fallback' }, 'Chart unavailable.'));
  });

Chart Configuration:

  • Type: Stacked area chart (line chart with fill)
  • Datasets: Value of investment (contributions) + Total interest earned
  • Colors: Light blue for contributions, dark blue for interest
  • Scales: X-axis shows months, Y-axis shows currency values with CAD formatting
  • Performance: Max 120 data points to keep chart rendering fast even with 50-year projections

Accessibility Features

  • ARIA Live Region: Output section announces results to screen readers when calculations update
  • Semantic HTML: Proper <form>, <label>, and <input> elements
  • Keyboard Navigation: All inputs accessible via keyboard (tab, arrow keys for select/number inputs)
  • Tooltips: Info icons with title attributes explain each field

Best Practices

  • Real-Time Feedback: Calculate on every input change for immediate feedback
  • Sensible Limits: Constrain years (0.5-50) and rates to prevent unrealistic projections
  • Chart Cleanup: Destroy previous Chart.js instance before creating new one to prevent memory leaks
  • Currency Formatting: Use Intl.NumberFormat for locale-aware currency display
  • Lazy Loading: Load Chart.js asynchronously to avoid blocking page load
  • Fallback Handling: Gracefully handle Chart.js load failures with fallback message
  • Responsive Design: Mobile-first with stacked layout, side-by-side on desktop (900px+)
  • User Guidance: Provide legend/disclaimer explaining calculator limitations
  • Performance: Limit chart data points to keep rendering fast

    export function createTag(tag, attributes = {}, children = []) {
  const el = document.createElement(tag);
  Object.entries(attributes).forEach(([key, value]) => {
    if (value !== undefined && value !== null) el.setAttribute(key, value);
  });
  if (typeof children === 'string') el.textContent = children;
  else if (Array.isArray(children)) el.append(...children);
  else if (children instanceof Node) el.append(children);
  return el;
}

export async function loadChartJs() {
  if (window.Chart) return window.Chart;
  const script = document.createElement('script');
  script.src = 'https://cdn.jsdelivr.net/npm/chart.js@4.4.0/dist/chart.umd.min.js';
  document.head.append(script);
  return new Promise((resolve, reject) => {
    script.onload = () => resolve(window.Chart);
    script.onerror = reject;
  });
}

export function createChart(canvas, config) {
  if (!window.Chart) return undefined;
  return new window.Chart(canvas, config);
}
