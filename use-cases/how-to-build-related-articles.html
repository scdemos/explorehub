How to Build a Related Articles Block

Dynamic content recommendations powered by query-index

Block Query Index Intermediate

Overview

The Related Articles block displays contextually relevant content based on keywords, tags, or manually curated links. It queries the site's query-index to find matching articles and presents them in either a horizontal slider or bento-style grid layout.

Key Features

  • Query-index powered - Dynamic content discovery
  • Keyword matching - Filter by metadata keywords/tags
  • Manual curation - Hand-picked article links
  • Two variants - Horizontal slider (default) or bento grid
  • Responsive carousel - Touch-friendly with navigation arrows
  • Exclusion filters - Remove unwanted articles
  • Random selection - Display random articles

// Links mode: extract  tags from block
function getLinksFromBlock(block) {
  const anchors = block.querySelectorAll('a[href]');
  return [...anchors].map(a => ({
    path: pathFromHref(a.href),
    title: a.textContent.trim()
  }));
}

// Dynamic mode: fetch query-index and match
const indexRows = await fetchQueryIndexAll();

Step 3: Match Articles by Keywords

Filter query-index rows by keyword matching (related-articles.js:36-60):

function matchArticles(rows, keywordsConfig, excludedConfig, limit) {
  const requested = parseKeywords(keywordsConfig);
  const excluded = parseKeywords(excludedConfig);
  const isRandom = requested.includes('random');

  // Filter by keywords
  const filtered = isRandom || !requested.length
    ? rows
    : rows.filter(row =>
        requested.some(keyword => rowMatchesKeyword(row, keyword))
      );

  // Remove excluded
  const withoutExcluded = excluded.length
    ? filtered.filter(row =>
        !excluded.some(keyword => rowMatchesKeyword(row, keyword))
      )
    : filtered;

  // Dedupe and sort
  const deduped = [...new Set(withoutExcluded.map(r => r.path))];

  // Shuffle if random or multiple keywords
  if (isRandom) return shuffle(deduped).slice(0, limit);

  return deduped
    .sort((a, b) => getContentTimestamp(b) - getContentTimestamp(a))
    .slice(0, limit);
}

Step 4: Build Card UI

Create card elements for slider or bento variant (related-articles.js:66-83):

function buildSliderCard(article) {
  const href = normalizePath(article.path);
  const link = createTag('a', {
    href,
    class: 'related-articles-card-link'
  });

  const content = createTag('div', {
    class: 'related-articles-card-content'
  });
  content.append(createTag('h3', {}, article.title || href));

  if (article.description) {
    content.append(createTag('p', {
      class: 'related-articles-card-description'
    }, article.description));
  }

  const date = article.date || article.publisheddate;
  if (date) {
    content.append(createTag('p', {
      class: 'related-articles-card-date'
    }, formatDate(date)));
  }

  link.append(content);
  return createTag('li', { class: 'related-articles-card' }, link);
}

Step 5: Add Slider Controls

For default variant, add prev/next navigation with scroll behavior (related-articles.js:93-101):

function updateSliderButtons(list, prevBtn, nextBtn) {
  const max = list.scrollWidth - list.clientWidth - 1;
  const atStart = list.scrollLeft <= 0;
  const atEnd = list.scrollLeft >= max;

  prevBtn.disabled = atStart;
  prevBtn.classList.toggle('is-hidden', atStart);
  nextBtn.disabled = atEnd;
  nextBtn.classList.toggle('is-hidden', atEnd);
}

// Smooth scroll on button click
prevBtn.addEventListener('click', () => {
  const step = getScrollStep(list);
  list.scrollBy({ left: -step, behavior: 'smooth' });
});

// Update button state on scroll
list.addEventListener('scroll', () => {
  updateSliderButtons(list, prevBtn, nextBtn);
});

// Modify rowMatchesKeyword() for fuzzy matching:
function rowMatchesKeyword(row, keyword) {
  const articleKeywords = parseKeywords(getArticleKeywords(row));
  return articleKeywords.some(ak =>
    ak.toLowerCase().includes(keyword.toLowerCase())
  );
}
